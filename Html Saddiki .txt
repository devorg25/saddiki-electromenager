<script>
  const SUPABASE_URL = "PUT_YOUR_SUPABASE_URL_HERE";
  const SUPABASE_ANON_KEY = "PUT_YOUR_SUPABASE_ANON_KEY_HERE";
  const supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

  let currentLanguage = localStorage.getItem("language") || "fr";

  function changeLanguage(language) {
    currentLanguage = language;
    localStorage.setItem("language", language);
    document.documentElement.lang = language;
    document.documentElement.dir = language === "ar" ? "rtl" : "ltr";

    document.querySelectorAll("[data-fr]").forEach(element => {
      const text = element.getAttribute(`data-${language}`);
      if (text) element.textContent = text;
    });

    document.getElementById("customerName").placeholder = language === "ar" ? "الاسم الكامل" : "Votre nom";
    document.getElementById("customerPhone").placeholder = "0772656995";
    document.getElementById("customerAddress").placeholder = language === "ar" ? "عنوان التوصيل" : "Adresse";
    document.getElementById("customerProduct").placeholder = language === "ar" ? "اسم المنتج" : "Produit";
    document.getElementById("customerMessage").placeholder = language === "ar" ? "تفاصيل إضافية..." : "Informations supplémentaires...";

    document.getElementById("frBtn").classList.toggle("active", language === "fr");
    document.getElementById("arBtn").classList.toggle("active", language === "ar");
  }

  async function loadProducts() {
    const { data, error } = await supabase
      .from("products")
      .select("*")
      .eq("is_active", true)
      .order("created_at", { ascending: false });

    const grid = document.getElementById("productsGrid");

    if (error) {
      grid.innerHTML = `<div class="empty-products"><h3>Error loading products</h3><p>${error.message}</p></div>`;
      return;
    }

    if (!data || data.length === 0) {
      grid.innerHTML = `
        <div class="empty-products">
          <h3 data-fr="Produits bientôt disponibles" data-ar="المنتجات ستكون متوفرة قريباً">Produits bientôt disponibles</h3>
          <p data-fr="Les produits ajoutés depuis le tableau de bord apparaîtront ici." data-ar="المنتجات التي تتم إضافتها من لوحة الإدارة ستظهر هنا.">
            Les produits ajoutés depuis le tableau de bord apparaîtront ici.
          </p>
        </div>`;
      changeLanguage(currentLanguage);
      return;
    }

    grid.innerHTML = data.map(product => `
      <div class="product-card">
        <img class="product-image" src="${product.image_url || 'https://via.placeholder.com/600x400?text=Saddiki'}" alt="${product.name_fr}">
        <div class="product-content">
          <h3>${currentLanguage === 'ar' ? product.name_ar : product.name_fr}</h3>
          <p class="product-description">${currentLanguage === 'ar' ? (product.description_ar || '') : (product.description_fr || '')}</p>
          <div class="product-bottom">
            <span class="price">${product.price} DA</span>
            <button class="btn btn-primary" type="button" onclick="prefillOrder('${currentLanguage === 'ar' ? product.name_ar : product.name_fr}')">
              ${currentLanguage === 'ar' ? 'اطلب الآن' : 'Commander'}
            </button>
          </div>
        </div>
      </div>
    `).join("");
  }

  function prefillOrder(productName) {
    document.getElementById("customerProduct").value = productName;
    document.getElementById("order").scrollIntoView({ behavior: "smooth" });
  }

  document.getElementById("orderForm").addEventListener("submit", async function (event) {
    event.preventDefault();

    const name = document.getElementById("customerName").value.trim();
    const phone = document.getElementById("customerPhone").value.trim();
    const address = document.getElementById("customerAddress").value.trim();
    const product = document.getElementById("customerProduct").value.trim();
    const quantity = document.getElementById("customerQuantity").value;
    const message = document.getElementById("customerMessage").value.trim();

    if (!name || !phone || !address || !product || !quantity) {
      alert(currentLanguage === "ar" ? "يرجى ملء جميع المعلومات المطلوبة." : "Veuillez remplir tous les champs obligatoires.");
      return;
    }

    const orderText = currentLanguage === "ar"
      ? `طلب جديد:
الاسم: ${name}
الهاتف: ${phone}
العنوان: ${address}
المنتج: ${product}
الكمية: ${quantity}
الرسالة: ${message || "-"}`
      : `Nouvelle commande:
Nom: ${name}
Téléphone: ${phone}
Adresse: ${address}
Produit: ${product}

  async function checkAdminSession() {
    const { data } = await supabase.auth.getSession();
    const session = data.session;

    if (!session) {
      document.getElementById("loginPanel").classList.remove("hidden");
      document.getElementById("dashboardPanel").classList.add("hidden");
      return;
    }

    const { data: roleData } = await supabase
      .from("user_roles")
      .select("role")
      .eq("user_id", session.user.id)
      .single();

    if (roleData && roleData.role === "admin") {
      document.getElementById("loginPanel").classList.add("hidden");
      document.getElementById("dashboardPanel").classList.remove("hidden");
    } else {
      await supabase.auth.signOut();
      document.getElementById("loginPanel").classList.remove("hidden");
      document.getElementById("dashboardPanel").classList.add("hidden");
    }
  }

  document.getElementById("adminLoginForm").addEventListener("submit", async (e) => {
    e.preventDefault();

    const email = document.getElementById("adminEmail").value.trim();
    const password = document.getElementById("adminPassword").value;

    const { data, error } = await supabase.auth.signInWithPassword({ email, password });

    if (error) {
      alert(error.message);
      return;
    }

    const { data: roleData, error: roleError } = await supabase
      .from("user_roles")
      .select("role")
      .eq("user_id", data.user.id)
      .single();

    if (roleError || !roleData || roleData.role !== "admin") {
      await supabase.auth.signOut();
      alert("Access denied");
      return;
    }

    await checkAdminSession();
  });
